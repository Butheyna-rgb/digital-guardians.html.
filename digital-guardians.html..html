let score=0;

let current=0;

let timeLeft=30;

let interval=null;

let answered=false;


const $=id=>document.getElementById(id);


function updateStats(){

 $("score").textContent=score;

 $("stageNum").textContent=current+" / 5";

 $("progress").style.width=(current/5*100)+"%";

 $("topTimer").textContent=timeLeft;

}


function startGame(){

 $("welcome").classList.add("hidden");

 $("result").classList.add("hidden");

 $("game").classList.remove("hidden");

 current=0;

 score=0;

 loadStage();

}


function loadStage(){

 clearInterval(interval);

 answered=false;

 timeLeft=30;

 const s=stages[current];


 $("stageLabel").textContent="المهمة "+["الأولى","الثانية","الثالثة","الرابعة","الخامسة"][current];

 $("title").textContent=s.title;

 $("story").textContent=s.story;

 $("timer").textContent=timeLeft;

 $("feedback").className="feedback";

 $("feedback").textContent="";

 $("next").classList.add("hidden");

 $("choices").innerHTML="";


 s.choices.forEach((choice,i)=>{

  const btn=document.createElement("button");

  btn.className="choice";

  btn.textContent=["أ","ب","ج"][i]+" — "+choice;

  btn.onclick=()=>choose(i);

  $("choices").appendChild(btn);

 });


 updateStats();

 interval=setInterval(()=>{

  if(answered)return;

  timeLeft--;

  $("timer").textContent=timeLeft;

  $("topTimer").textContent=timeLeft;

  if(timeLeft<=0){

   clearInterval(interval);

   timeUp();

  }

 },1000);

}


function choose(index){

 if(answered)return;

 answered=true;

 clearInterval(interval);


 const s=stages[current];

 const buttons=$("choices").querySelectorAll("button");

 buttons.forEach((btn,i)=>{

  btn.disabled=true;

  if(i===s.correct)btn.classList.add("correct");

  else if(i===index)btn.classList.add("wrong");

 });


 if(index===s.correct){

  score+=10;

  showFeedback(true,"🌟 إجابة صحيحة! +10 نقاط. ",s.reason);

 }else{

  showFeedback(false,"💡 ليست الإجابة الأكثر أماناً. ",s.reason);

 }

 finishStage();

}


function timeUp(){

 if(answered)return;

 answered=true;

 const s=stages[current];

 const buttons=$("choices").querySelectorAll("button");

 buttons.forEach((btn,i)=>{

  btn.disabled=true;

  if(i===s.correct)btn.classList.add("correct");

 });

 showFeedback(false,"⏰ انتهى الوقت! لم تحصلي على نقاط هذه المرحلة. ",s.reason);

 finishStage();

}


function showFeedback(good,heading,reason){

 const box=$("feedback");

 box.className="feedback "+(good?"good":"bad");

 box.innerHTML="<strong>"+heading+"</strong><br>"+reason;

}


function finishStage(){

 updateStats();

 $("next").classList.remove("hidden");

 $("next").textContent=current===4?"🏆 عرض النتيجة":"المرحلة التالية ←";

}


function nextStage(){

 current++;

 $("stageNum").textContent=current+" / 5";

 $("progress").style.width=(current/5*100)+"%";

 if(current>=5){

  showResult();

 }else{

  loadStage();

 }

}


function showResult(){

 clearInterval(interval);

 $("game").classList.add("hidden");

 $("result").classList.remove("hidden");

 $("finalScore").textContent=score+" / 50";


 if(score>=40){

  $("endIcon").textContent="🏆👑";

  $("endTitle").textContent="حارسة العالم الرقمي!";

  $("endMessage").textContent="أبدعتِ! أنتِ حارسة واعية تعرف كيف تحمي نفسها والآخرين في العالم الرقمي. واصلي التميز!";

 }else if(score>=20){

  $("endIcon").textContent="🌱💜";

  $("endTitle").textContent="خطوات نحو الأمان";

  $("endMessage").textContent="أحسنتِ المحاولة! لديكِ معرفة جيدة، وتحتاجين إلى مراجعة بعض قواعد الاستخدام الآمن للتقنية. أعيدي المغامرة لتصبحي أقوى!";

 }else{

  $("endIcon").textContent="⚠️🛡️";

  $("endTitle").textContent="رحلتكِ نحو الأمان مستمرة";

  $("endMessage").textContent="لا تستسلمي! تعلّمي قواعد حماية الحسابات والخصوصية والإبلاغ، ثم أعيدي اللعب. كل محاولة تقربكِ من الأمان الرقمي.";

 }

 $("stageNum").textContent="5 / 5";

 $("topTimer").textContent="0";

 $("progress").style.width="100%";

}


function restart(){

 $("result").classList.add("hidden");

 $("welcome").classList.remove("hidden");

 $("progress").style.width="0";

 $("stageNum").textContent="0 / 5";

 $("topTimer").textContent="30";

 $("score").textContent="0";
